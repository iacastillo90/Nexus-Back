```dart
import 'package:riverpod_annotation/riverpod_annotation.dart';
import '../../domain/entities/user_entity.dart';
import '../../domain/repositories/auth_repository.dart';
import '../../../core/storage/secure_storage_service.dart';

part 'auth_controller.g.dart';

@Riverpod(keepAlive: true)
class AuthController extends _$AuthController {
  @override
  Future<UserEntity?> build() async {
    // Intentar recuperar usuario del token almacenado
    final token = await ref.read(secureStorageServiceProvider).getToken();
    
    if (token == null) return null;
    
    // Validar token con el backend
    try {
      final user = await ref.read(authRepositoryProvider).getCurrentUser();
      
      // Conectar socket con el token
      ref.read(socketServiceProvider).connect(token);
      
      return user;
    } catch (e) {
      // Token inválido, limpiar storage
      await ref.read(secureStorageServiceProvider).deleteToken();
      return null;
    }
  }
  
  Future<void> login(String email, String password) async {
    state = const AsyncLoading();
    
    state = await AsyncValue.guard(() async {
      final authResponse = await ref.read(authRepositoryProvider).login(
        email,
        password,
      );
      
      // Guardar token
      await ref.read(secureStorageServiceProvider).saveToken(
        authResponse.token,
      );
      
      // Conectar socket
      ref.read(socketServiceProvider).connect(authResponse.token);
      
      return authResponse.user;
    });
  }
  
  Future<void> register({
    required String username,
    required String email,
    required String password,
  }) async {
    state = const AsyncLoading();
    
    state = await AsyncValue.guard(() async {
      final authResponse = await ref.read(authRepositoryProvider).register(
        username: username,
        email: email,
        password: password,
      );
      
      await ref.read(secureStorageServiceProvider).saveToken(
        authResponse.token,
      );
      
      ref.read(socketServiceProvider).connect(authResponse.token);
      
      return authResponse.user;
    });
  }
  
  Future<void> logout() async {
    // Desconectar socket
    ref.read(socketServiceProvider).disconnect();
    
    // Borrar token
    await ref.read(secureStorageServiceProvider).deleteToken();
    
    // Limpiar estado
    state = const AsyncData(null);
    
    // Invalidar todos los providers dependientes
    ref.invalidate(feedControllerProvider);
    ref.invalidate(profileControllerProvider);
    ref.invalidate(notificationControllerProvider);
  }
  
  Future<void> updateProfile({
    String? firstName,
    String? lastName,
    String? bio,
  }) async {
    final currentUser = state.value;
    if (currentUser == null) return;
    
    state = const AsyncLoading();
    
    state = await AsyncValue.guard(() async {
      final updatedUser = await ref.read(authRepositoryProvider).updateUser(
        userId: currentUser.id,
        firstName: firstName,
        lastName: lastName,
      );
      
      return updatedUser;
    });
  }
}
```

```dart
import 'package:riverpod_annotation/riverpod_annotation.dart';
import 'package:socket_io_client/socket_io_client.dart' as IO;
import '../../../../core/constants/app_constants.dart';

part 'socket_service_provider.g.dart';

@Riverpod(keepAlive: true)
SocketService socketService(SocketServiceRef ref) {
  final service = SocketService(ref);
  
  // Cleanup al dispose
  ref.onDispose(() {
    service.disconnect();
  });
  
  return service;
}

class SocketService {
  final Ref _ref;
  IO.Socket? _socket;
  
  SocketService(this._ref);
  
  void connect(String token) {
    if (_socket != null && _socket!.connected) return;
    
    _socket = IO.io(
      AppConstants.apiBaseUrl,
      IO.OptionBuilder()
          .setTransports(['websocket'])
          .setAuth({'token': token})
          .disableAutoConnect()
          .build(),
    );
    
    _socket!.connect();
    
    _socket!.onConnect((_) {
      print('✅ [Socket] Connected');
      _setupListeners();
    });
    
    _socket!.onConnectError((data) {
      print('❌ [Socket] Connection Error: $data');
    });
    
    _socket!.onDisconnect((_) {
      print('🔌 [Socket] Disconnected');
    });
  }
  
  void _setupListeners() {
    // Notificación nueva
    _socket!.on('notification:new', (data) {
      print('🔔 [Socket] Nueva notificación: $data');
      _ref.read(notificationControllerProvider.notifier).addNotification(data);
    });
    
    // Actualización de Karma
    _socket!.on('karma:updated', (data) {
      print('⭐ [Socket] Karma actualizado: $data');
      _ref.invalidate(karmaControllerProvider);
    });
    
    // Nuevo mensaje (Chat)
    _socket!.on('message:new', (data) {
      print('💬 [Socket] Nuevo mensaje: $data');
      // TODO: Actualizar chat controller correspondiente
    });
    
    // Dream actualizado
    _socket!.on('dream:updated', (data) {
      print('🌙 [Socket] Dream actualizado: $data');
      // TODO: Invalidar dream controller
    });
    
    // Usuario online/offline
    _socket!.on('user:online', (data) {
      print('🟢 [Socket] Usuario online: $data');
    });
    
    _socket!.on('user:offline', (data) {
      print('⚪ [Socket] Usuario offline: $data');
    });
  }
  
  void disconnect() {
    _socket?.disconnect();
    _socket = null;
  }
  
  void emit(String event, dynamic data) {
    if (_socket == null || !_socket!.connected) {
      print('⚠️ [Socket] Cannot emit - not connected');
      return;
    }
    _socket!.emit(event, data);
  }
  
  bool get isConnected => _socket != null && _socket!.connected;
}
```

```dart
import 'package:riverpod_annotation/riverpod_annotation.dart';

part 'location_controller.g.dart';

@Riverpod(keepAlive: true)
class LocationController extends _$LocationController {
  @override
  Stream<Position> build() {
    return Geolocator.getPositionStream(
      locationSettings: const LocationSettings(
        accuracy: LocationAccuracy.high,
        distanceFilter: 10, // Actualizar cada 10 metros
      ),
    );
  }
  
  Future<bool> requestPermission() async {
    bool serviceEnabled = await Geolocator.isLocationServiceEnabled();
    if (!serviceEnabled) {
      return false;
    }
    
    LocationPermission permission = await Geolocator.checkPermission();
    if (permission == LocationPermission.denied) {
      permission = await Geolocator.requestPermission();
      if (permission == LocationPermission.denied) {
        return false;
      }
    }
    
    if (permission == LocationPermission.deniedForever) {
      return false;
    }
    
    return true;
  }
  
  Future<Position?> getCurrentPosition() async {
    try {
      return await Geolocator.getCurrentPosition(
        desiredAccuracy: LocationAccuracy.high,
      );
    } catch (e) {
      print('Error getting position: $e');
      return null;
    }
  }
}
```

## 6.2 Providers Efímeros (AutoDispose - Se destruyen al salir)

### FeedController (lib/features/feed/presentation/providers/feed_controller.dart)

```dart
import 'package:riverpod_annotation/riverpod_annotation.dart';
import '../../domain/entities/post_entity.dart';
import '../../domain/repositories/feed_repository.dart';

part 'feed_controller.g.dart';

@riverpod
class FeedController extends _$FeedController {
  int _currentPage = 1;
  static const _pageSize = 20;
  bool _hasMore = true;
  
  @override
  Future<List<PostEntity>> build() async {
    // Intentar cargar del caché local primero
    final cachedPosts = await ref.read(feedRepositoryProvider).getCachedFeed();
    
    if (cachedPosts.isNotEmpty) {
      // Devolver caché inmediatamente
      state = AsyncData(cachedPosts);
      
      // Refrescar en background
      _refreshFeed();
      
      return cachedPosts;
    }
    
    // Si no hay caché, cargar desde API
    return _loadFeed();
  }
  
  Future<List<PostEntity>> _loadFeed() async {
    final posts = await ref.read(feedRepositoryProvider).getFeed(
      page: _currentPage,
      limit: _pageSize,
    );
    
    _hasMore = posts.length == _pageSize;
    
    // Guardar en caché
    await ref.read(feedRepositoryProvider).cacheFeed(posts);
    
    return posts;
  }
  
  Future<void> _refreshFeed() async {
    try {
      final freshPosts = await ref.read(feedRepositoryProvider).getFeed(
        page: 1,
        limit: _pageSize,
      );
      
      await ref.read(feedRepositoryProvider).cacheFeed(freshPosts);
      
      state = AsyncData(freshPosts);
      _currentPage = 1;
      _hasMore = freshPosts.length == _pageSize;
    } catch (e) {
      // No actualizar estado si falla el refresh
      print('Error refreshing feed: $e');
    }
  }
  
  Future<void> loadMore() async {
    if (!_hasMore || state.isLoading) return;
    
    final currentPosts = state.value ?? [];
    if (currentPosts.isEmpty) return;
    
    _currentPage++;
    
    try {
      final newPosts = await ref.read(feedRepositoryProvider).getFeed(
        page: _currentPage,
        limit: _pageSize,
      );
      
      _hasMore = newPosts.length == _pageSize;
      
      state = AsyncData([...currentPosts, ...newPosts]);
    } catch (e) {
      _currentPage--; // Revertir si falla
      rethrow;
    }
  }
  
  Future<void> refresh() async {
    _currentPage = 1;
    _hasMore = true;
    state = const AsyncLoading();
    state = await AsyncValue.guard(() => _loadFeed());
  }
  
  void updatePostLocally(String postId, PostEntity Function(PostEntity) update) {
    state = state.whenData((posts) {
      return posts.map((post) {
        if (post.id == postId) {
          return update(post);
        }
        return post;
      }).toList();
    });
  }
  
  void removePostLocally(String postId) {
    state = state.whenData((posts) {
      return posts.where((post) => post.id != postId).toList();
```

```dart
@riverpod
class ChatController extends _$ChatController {
  @override
  Future<List<MessageEntity>> build(String mentorId) async {
    // Cargar historial del chat
    final messages = await ref.read(chatRepositoryProvider).getMessages(
      mentorId: mentorId,
      limit: 50,
    );
    
    // Escuchar nuevos mensajes por socket
    _setupSocketListener();
    
    return messages.reversed.toList(); // Más recientes al final
  }
  
  void _setupSocketListener() {
    ref.read(socketServiceProvider).on('message:new', (data) {
      final message = MessageEntity.fromJson(data);
      
      // Solo agregar si es del mentor correcto
      if (message.senderId == mentorId || message.recipientId == mentorId) {
        _addMessageToState(message);
      }
    });
  }
  
  void _addMessageToState(MessageEntity message) {
    state = state.whenData((messages) {
      return [...messages, message];
    });
  }
  
  Future<void> sendMessage(String content) async {
    // Crear mensaje temporal (optimistic UI)
    final tempMessage = MessageEntity(
      id: 'temp_${DateTime.now().millisecondsSinceEpoch}',
      senderId: ref.read(authControllerProvider).value!.id,
      recipientId: mentorId,
      content: content,
      type: MessageType.text,
      createdAt: DateTime.now(),
      isSent: false,
    );
    
    // Agregar a la lista localmente
    _addMessageToState(tempMessage);
    
    try {
      // Enviar al backend
      final sentMessage = await ref.read(chatRepositoryProvider).sendMessage(
        mentorId: mentorId,
        content: content,
      );
      
      // Reemplazar mensaje temporal con el real
      state = state.whenData((messages) {
        return messages.map((msg) {
          if (msg.id == tempMessage.id) {
            return sentMessage.copyWith(isSent: true);
          }
          return msg;
        }).toList();
      });
      
      HapticFeedback.lightImpact();
    } catch (e) {
      // Marcar como error
      state = state.whenData((messages) {
        return messages.map((msg) {
          if (msg.id == tempMessage.id) {
            return msg.copyWith(isSent: false);
          }
          return msg;
        }).toList();
      });
      
      ref.read(errorHandlerProvider).showError(e);
    }
  }
  
  Future<void> loadMoreMessages() async {
    final currentMessages = state.value ?? [];
    if (currentMessages.isEmpty) return;
    
    final oldestMessageTime = currentMessages.first.createdAt;
    
    try {
      final olderMessages = await ref.read(chatRepositoryProvider).getMessages(
        mentorId: mentorId,
        before: oldestMessageTime,
        limit: 20,
      );
      
      state = AsyncData([...olderMessages.reversed.toList(), ...currentMessages]);
    } catch (e) {
      print('Error loading more messages: $e');
    }
  }
}
```

```dart
class MainShell extends StatelessWidget {
  final Widget child;
  final String currentLocation;
  
  const MainShell({
    required this.child,
    required this.currentLocation,
    super.key,
  });
  
  int _getCurrentIndex() {
    if (currentLocation.startsWith('/reality')) return 1;
    if (currentLocation.startsWith('/create')) return 2;
    if (currentLocation.startsWith('/chat')) return 3;
    if (currentLocation.startsWith('/profile')) return 4;
    return 0; // Feed por defecto
  }
  
  void _onItemTapped(BuildContext context, int index) {
    switch (index) {
      case 0:
        context.go('/');
        break;
      case 1:
        context.go('/reality');
        break;
      case 2:
        context.go('/create');
        break;
      case 3:
        context.go('/chat');
        break;
      case 4:
        context.go('/profile');
        break;
    }
  }
```
